cmake_minimum_required(VERSION 3.14)
project(dwm_dashboard C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(FetchContent)
set(CUI_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CUI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CUI_TEXT_SHAPING ON CACHE BOOL "" FORCE)
set(CUI_PLATFORM x11 CACHE STRING "" FORCE)
FetchContent_Declare(cui
  GIT_REPOSITORY https://github.com/yappologistic/CUI.git
  GIT_TAG 7bb08ff7c56eb912fe7133985f8b7fa65098ea2e)
FetchContent_MakeAvailable(cui)
find_package(Threads REQUIRED)
add_executable(dwm-dashboard dashboard.c)
add_executable(dwm-launcher launcher.c)
foreach(tool dwm-dashboard dwm-launcher)
  target_compile_definitions(${tool} PRIVATE _POSIX_C_SOURCE=200809L)
  target_compile_options(${tool} PRIVATE -Wall -Wextra)
  target_link_libraries(${tool} PRIVATE cui::cui Threads::Threads)
endforeach()
find_package(X11 REQUIRED)
if(NOT X11_Xext_FOUND)
  message(FATAL_ERROR "The launcher requires the X11 Shape extension (libXext)")
endif()
target_include_directories(dwm-launcher PRIVATE ${X11_Xext_INCLUDE_PATH})
target_link_libraries(dwm-launcher PRIVATE ${X11_Xext_LIB})
install(TARGETS dwm-dashboard dwm-launcher RUNTIME DESTINATION bin)
