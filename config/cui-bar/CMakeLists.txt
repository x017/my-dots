cmake_minimum_required(VERSION 3.14)
project(cui_bar C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(FetchContent)
set(CUI_WITH_APP OFF CACHE BOOL "" FORCE)
set(CUI_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CUI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CUI_STB_IMAGE ON CACHE BOOL "" FORCE)
FetchContent_Declare(cui GIT_REPOSITORY https://github.com/yappologistic/CUI.git
  GIT_TAG 7bb08ff7c56eb912fe7133985f8b7fa65098ea2e)
FetchContent_MakeAvailable(cui)
find_package(X11 REQUIRED)
find_package(Threads REQUIRED)
add_executable(cui-bar bar.c)
target_compile_definitions(cui-bar PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(cui-bar PRIVATE -Wall -Wextra -ffunction-sections -fdata-sections)
target_include_directories(cui-bar PRIVATE ${X11_INCLUDE_DIR})
target_link_libraries(cui-bar PRIVATE cui::cui ${X11_LIBRARIES} Threads::Threads ${CMAKE_DL_LIBS})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(cui-bar PRIVATE -Wl,--gc-sections)
endif()
install(TARGETS cui-bar RUNTIME DESTINATION bin)
